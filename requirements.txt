# ⚙️ TrioForge dependencies — lightweight core
#
# This is everything needed to RUN TrioForge. The heavy ML stack
# (sentence-transformers + torch + CUDA, ~2 GB) is OPTIONAL and lives in
# requirements-ml.txt — see the bottom of this file.

# ── Core web app ────────────────────────────────────────────────
flask>=3.0.0
flask-compress>=1.13
requests>=2.31.0

# ── Servers ─────────────────────────────────────────────────────
waitress==3.0.2          # Windows / macOS HTTPS server
gunicorn==26.0.0         # Linux / Docker server

# ── LLM providers ───────────────────────────────────────────────
groq>=0.3.0              # Groq provider (SDK)
huggingface_hub>=0.16.0  # Hugging Face provider
google-genai>=2.0.0      # Gemini provider (chat + image + video)

# ── File / PDF parsing ──────────────────────────────────────────
pypdf>=3.0.0             # PDF text extraction (uploaded PDFs)

# ── Notes (Obsidian-style frontmatter) ──────────────────────────
# python-frontmatter, NOT the older `frontmatter` package: that one hard-pins
# PyYAML==5.1, a 2019 release with no wheels for Python 3.12+, so a fresh install
# has to COMPILE it and fails on any machine without the Visual C++ build tools.
# Same import name (`import frontmatter`) and same API - loads / Post / dumps.
python-frontmatter>=1.0.0

# ── Performance / utilities ─────────────────────────────────────
orjson>=3.9.0            # faster JSON (falls back to std json)
psutil>=5.9.0            # RAM / VRAM live monitor
pynvml>=11.525.0         # NVIDIA GPU monitoring
duckduckgo-search>=6.0.0 # optional web search

# ── Memory vault (DuckDB + Bloom filter gate) ───────────────────
# py/memory.py keeps durable facts in a DuckDB file with a Bloom filter held
# in RAM in front of it, so a key that is definitely absent costs no disk read.
# It is imported lazily: with duckdb missing, /memory and the agent's memory
# tool report how to install it instead of breaking the app.
duckdb>=1.1.0

# ── Terminal client (`forge` / `trioforge`) ─────────────────────
# These used to be installed separately by install.sh, which meant a fresh clone
# that ran run.sh (or pip install -r requirements.txt) got the web app but
# `forge` died on ModuleNotFoundError. One manifest now covers both.
rich>=13.7.0
prompt_toolkit>=3.0.43
textual>=0.80.0
# The terminal client's model transport (py/tui/backend.py). It used to arrive
# only as a side effect of groq/google-genai, so a change there could drop it.
httpx>=0.27.0

# ── Semantic search / embeddings (OPTIONAL — app runs without) ──
# These pull in torch + the CUDA toolkit (~2 GB), so they are NOT installed by
# default. If you want semantic RAG (meaning-based search instead of keywords),
# install them explicitly:
#
#     python -m pip install -r requirements-ml.txt
#
# The app auto-detects them; without them, document chat uses a fast keyword
# fallback. See also requirements-ml.txt.
