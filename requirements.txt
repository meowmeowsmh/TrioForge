# ⚙️ TrioForge dependencies — lightweight core
#
# This is everything needed to RUN TrioForge. The heavy ML stack
# (sentence-transformers + torch + CUDA, ~2 GB) is OPTIONAL and lives in
# requirements-ml.txt — see the bottom of this file.

# Version floors are the releases this project is TESTED against (the ones the
# venv and the CI matrix install), and every one carries an upper bound. Before
# this they were open `>=` floors, so a fresh install in six months could resolve
# to a major version nobody had ever run — the kind of break that looks like a
# TrioForge bug. `>=floor,<next-major` keeps installs reproducible without pinning
# so hard that a security patch cannot come in.
#
# ── Core web app ────────────────────────────────────────────────
flask>=3.1.3,<4.0.0
# Security floor for a transitive dep: flask carries werkzeug, and an open
# resolve could land on the pre-CVE-2026-102598 release. Held explicit so a
# fresh install cannot drift below the patched version.
werkzeug>=3.1.9,<4.0.0
flask-compress>=1.25,<2.0.0
requests>=2.34.2,<3.0.0
# Security floor for a transitive dep: requests carries urllib3, and an open
# resolve could land on a pre-CVE release. Held explicit so a fresh install
# cannot drift below the patched version.
urllib3>=2.8.0,<3.0.0

# ── Servers ─────────────────────────────────────────────────────
waitress==3.0.2          # Windows / macOS HTTPS server (exact: server behaviour)
gunicorn==26.0.0         # Linux / Docker server (exact: server behaviour)

# ── LLM providers ───────────────────────────────────────────────
groq>=1.7.0,<2.0.0       # Groq provider (SDK)
# <2 is deliberate. The OPTIONAL semantic stack (requirements-ml.txt) pins
# huggingface-hub<2, so an uncapped floor here lets the launch-time auto-upgrade
# replace 1.x with 2.x and quietly break sentence-transformers. The app only uses
# hf_hub_download/HfApi, which are unchanged in 1.x.
huggingface_hub>=0.16.0,<2.0.0  # Hugging Face provider
google-genai>=2.29.0,<3.0.0   # Gemini provider (chat + image + video)

# ── File / PDF parsing ──────────────────────────────────────────
pypdf>=6.20.0,<7.0.0     # PDF text extraction (uploaded PDFs)

# ── Notes (Obsidian-style frontmatter) ──────────────────────────
# python-frontmatter, NOT the older `frontmatter` package: that one hard-pins
# PyYAML==5.1, a 2019 release with no wheels for Python 3.12+, so a fresh install
# has to COMPILE it and fails on any machine without the Visual C++ build tools.
# Same import name (`import frontmatter`) and same API - loads / Post / dumps.
python-frontmatter>=1.3.0,<2.0.0

# ── Performance / utilities ─────────────────────────────────────
orjson>=3.13.0,<4.0.0    # faster JSON (falls back to std json)
psutil>=7.2.2,<8.0.0     # RAM / VRAM live monitor
pynvml>=13.0.1,<14.0.0   # NVIDIA GPU monitoring
duckduckgo-search>=8.1.1,<9.0.0  # optional web search

# ── Voice output (neural voices for the read-aloud button) ──────
# Windows only installs three SAPI voices (David, Mark, Zira) and the browser can
# only speak with what the OS has, so the voice picker offered exactly three.
# py/neural_voice.py adds the neural voices through edge-tts. The import is LAZY:
# without this package the app still runs, and the picker lists the local voices
# with the reason the neural list is missing. TRIOFORGE_TTS=off switches the
# network voices off entirely for offline-first installs.
edge-tts>=7.2.7,<8.0.0

# ── Memory vault (DuckDB + Bloom filter gate) ───────────────────
# py/memory.py keeps durable facts in a DuckDB file with a Bloom filter held
# in RAM in front of it, so a key that is definitely absent costs no disk read.
# It is imported lazily: with duckdb missing, /memory and the agent's memory
# tool report how to install it instead of breaking the app.
duckdb>=1.5.6,<2.0.0

# ── Terminal client (`forge` / `trioforge`) ─────────────────────
# These used to be installed separately by install.sh, which meant a fresh clone
# that ran run.sh (or pip install -r requirements.txt) got the web app but
# `forge` died on ModuleNotFoundError. One manifest now covers both.
rich>=15.0.0,<16.0.0
prompt_toolkit>=3.0.53,<4.0.0
textual>=8.2.8,<9.0.0
# The terminal client's model transport (py/tui/backend.py). It used to arrive
# only as a side effect of groq/google-genai, so a change there could drop it.
httpx>=0.28.1,<1.0.0

# ── Semantic search / embeddings (OPTIONAL — app runs without) ──
# These pull in torch + the CUDA toolkit (~2 GB), so they are NOT installed by
# default. If you want semantic RAG (meaning-based search instead of keywords),
# install them explicitly:
#
#     python -m pip install -r requirements-ml.txt
#
# The app auto-detects them; without them, document chat uses a fast keyword
# fallback. See also requirements-ml.txt.
